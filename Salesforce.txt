public class SelectionSort {

    public static List<Integer> selectionSort(List<Integer> inputList) {
        Integer n = inputList.size();

        for (Integer i = 0; i < n - 1; i++) {
            Integer minIndex = i;

            for (Integer j = i + 1; j < n; j++) {
                if (inputList[j] < inputList[minIndex]) {
                    minIndex = j;
                }
            }

            if (minIndex != i) {
                Integer temp = inputList[i];
                inputList[i] = inputList[minIndex];
                inputList[minIndex] = temp;
            }
        }
        return inputList;
    }
}
